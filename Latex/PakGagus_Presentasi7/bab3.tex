\section{Validasi Eksak \& Performa VQE}

\begin{frame}[fragile]{Algoritma Exhaustive Search (Brute Force)}
    \begin{itemize}
        \item \textbf{Tujuan:} Menemukan solusi eksak $\min_{x \in \{0,1\}^N} H(x)$.
    \end{itemize}
    \begin{block}{Pseudocode: Ising Ground State Search}
    \begin{enumerate}
        \item \textbf{Input:} Matriks interaksi $J_{ij}$, vektor bias $h_i$, konstanta $C$.
        \item \textbf{Initialize:} $E_{min} = \infty$, $x_{best} = \text{None}$.
        \item \textbf{Loop:} Untuk setiap bitstring $x \in \{0, 1\}^N$:
        \begin{itemize}
            \item Hitung Energi: $E(x) = \sum_{i<j} J_{ij} x_i x_j + \sum_{i} h_i x_i + C$
            \item \textbf{Update:} Jika $E(x) < E_{min}$:
            \begin{itemize}
                \item $E_{min} = E(x)$
                \item $x_{best} = x$
            \end{itemize}
        \end{itemize}
        \item \textbf{Output:} Energi Ground State $E_{min}$ dan konfigurasi $x_{best}$.
    \end{enumerate}
    \end{block}
\end{frame}

\begin{frame}{Validasi Brute Force (Exhaustive Search)}
    \begin{itemize}
        \item \textbf{Metodologi:} Menggunakan algoritma di atas pada ruang konfigurasi $2^N$.
        \item \textbf{Integrasi Grafik:} Hasil Brute Force diplot sebagai garis referensi horizontal pada grafik iterasi VQE.
        \item \textbf{Tujuan:} Memberikan standar kebenaran (\textit{ground truth}) untuk memastikan algoritma VQE tidak terjebak dalam lokal minimum yang tidak optimal.
    \end{itemize}
\end{frame}

\begin{frame}{Analisis Probabilitas Bitstring}
    \begin{itemize}
        \item \textbf{Depth Optimal:} Distribusi probabilitas diambil pada sirkuit dengan \textit{depth} yang menunjukkan konvergensi energi terbaik terhadap Brute Force.
        \item \textbf{Akurasi State:} Bitstring dengan probabilitas tertinggi ($|x\rangle$) secara konsisten sesuai dengan konfigurasi portofolio optimal dari hasil pencarian eksak.
        \item \textbf{Efisiensi:} Menunjukkan bahwa dengan jumlah qubit yang terbatas, VQE mampu mengisolasi solusi ground state di tengah ribuan kemungkinan konfigurasi.
    \end{itemize}
\end{frame}
