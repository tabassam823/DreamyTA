\chapter{Penurunan Rumus Fungsi Potensial dari Model Mean-Variance}
\label{appendix:derivation_epg}

Optimasi portofolio Markowitz bertujuan untuk memaksimalkan utilitas investor dengan menyeimbangkan antara ekspektasi imbal hasil dan risiko (varians). Fungsi Lagrangian dasar dari permasalahan ini dapat dituliskan sebagai:
\begin{equation}
    \mathcal{L}(\vec{\omega}) = \vec{\omega}^{T} \Sigma \vec{\omega} - \lambda(\mu^T\vec{\omega})
\end{equation}
di mana $\vec{\omega}$ adalah vektor bobot portofolio kontinu, $\Sigma$ adalah matriks kovariansi antar aset, $\mu$ adalah vektor ekspektasi imbal hasil, dan $\lambda$ adalah parameter yang berkaitan dengan penghindaran risiko (\textit{risk aversion}).

Dalam format matriks eksplisit, persamaan di atas menjadi:
\begin{equation}
    \mathcal{L(\vec{\omega})} = \begin{pmatrix}\omega_1 & \dots & \omega_n\end{pmatrix} 
    \begin{pmatrix}
    \sigma_{11} & \dots & \sigma_{1n} \\ 
    \vdots & \ddots & \vdots \\ 
    \sigma_{n1} & \dots & \sigma_{nn}
    \end{pmatrix} 
    \begin{pmatrix}\omega_1 \\ \vdots \\ \omega_n\end{pmatrix} 
    - \lambda \begin{pmatrix}\mu_1 & \dots & \mu_n\end{pmatrix} 
    \begin{pmatrix}\omega_1 \\ \vdots \\ \omega_n\end{pmatrix}
\end{equation}

Untuk mengonversi permasalahan kontinu ini ke dalam kerangka Teori Permainan, maka didefinisikan variabel keputusan diskrit $x_i \in \{0, 1\}$. Jika total aset yang diperbolehkan untuk dipilih adalah $k$, maka bobot diasumsikan sama rata, yaitu $\omega_i = \frac{x_i}{k}$. Substitusi ini mengubah fungsi Lagrangian menjadi:
\begin{equation}
    \mathcal{L}(\vec{x}) = \frac{\vec{x}^T}{k}\Sigma \frac{\vec{x}^T}{k} - \lambda \left(\mu^T\frac{\vec{x}}{k}\right)
\end{equation}
\begin{equation}
    \mathcal{L(\vec{x})} = \frac{1}{k^2}\begin{pmatrix} x_1 & \dots & x_n\end{pmatrix} 
    \Sigma
    \begin{pmatrix}x_1 \\ \vdots \\ x_n\end{pmatrix} 
    - \frac{\lambda}{k} \begin{pmatrix}\mu_1 & \dots & \mu_n\end{pmatrix} 
    \begin{pmatrix}x_1 \\ \vdots \\ x_n\end{pmatrix}
\end{equation}

Fungsi di atas merupakan fungsi minimalisasi risiko (biaya). Untuk mengubahnya menjadi fungsi maksimisasi utilitas $U(\vec{x})$, kita kalikan dengan $-\frac{1}{\lambda}$:
\begin{equation}
\begin{split}
    U(\vec{x}) &= -\frac{1}{\lambda} \mathcal{L}(\vec{x}) \\
    &= \left(\sum_{i=1}^N \mu_i \frac{x_i}{k}\right) - \frac{1}{\lambda k^2}\left(\sum_{i=1}^N\sum_{j=1}^N \sigma_{ij} x_i x_j\right)
\end{split}
\end{equation}
Mengingat bahwa toleransi risiko berbanding terbalik dengan penghindaran risiko, maka parameter $\gamma$ dapat didefinisikan dengan perbandingan  $\frac{1}{\lambda} \propto \frac{\gamma}{2}$. Dengan demikian, fungsi utilitas menjadi:
\begin{equation}
    U(\vec{x}) = \left(\sum_{i=1}^N \mu_i \frac{x_i}{k}\right) - \frac{\gamma}{2}\left(\sum_{i=1}^N\sum_{j=1}^N \sigma_{ij} \frac{x_i x_j}{k^2}\right)
\end{equation}

Untuk membuktikan bahwa sistem ini merupakan \textit{Exact Potential Game}, fungsi potensial global $\Phi(\vec{x})$ didefinisikan identik dengan fungsi utilitas agregat:
\begin{equation}
    \Phi(\vec{x}) = \sum_{l=1}^N \mu_l \frac{x_l}{k} - \frac{\gamma}{2}\sum_{i=1}^N\sum_{j=1}^N \sigma_{ij} \frac{x_i x_j}{k^2}
\end{equation}

Ekstraksi pengaruh satu agen (aset ke-$i$) terhadap fungsi potensial dilakukan dengan memisahkan indeks $i$ dari jumlahan linier (suku pertama) dan jumlahan kuadratik (suku kedua):
\begin{equation}
    \sum_{l=1}^N \mu_l \frac{x_l}{k} = \mu_i \frac{x_i}{k} + \sum_{l \neq i}^N \mu_l \frac{x_l}{k}
\end{equation}
\begin{equation}
    \sum_{i=1}^N\sum_{j=1}^N \sigma_{ij} \frac{x_i x_j}{k^2} = \sigma_i^2 \frac{x_i^2}{k^2} + 2\sum_{j \neq i}\sigma_{ij} \frac{x_ix_j}{k^2} + \sum_{l \neq i}\sum_{j \neq i} \sigma_{lj}\frac{x_l x_j}{k^2}
\end{equation}

Substitusi kembali ke persamaan $\Phi(\vec{x})$ dan dengan mempertimbangkan bahwa untuk variabel biner berlaku sifat indempoten $x_i^2 = x_i$, kita peroleh bagian fungsi potensial yang spesifik bergantung pada agen ke-$i$:
\begin{equation}
\begin{split}
    \Phi(\vec{x}) &= \mu_i \frac{x_i}{k} - \frac{\gamma}{2}\left(\sigma_i^2 \frac{x_i^2}{k^2} + 2\sum_{j \neq i}\sigma_{ij} \frac{x_ix_j}{k^2} \right) + \text{Suku tanpa } x_i \\
    &= \mu_i \frac{x_i}{k} - \frac{\gamma}{2}\sigma_i^2 \frac{x_i}{k^2} - \gamma \sum_{j \neq i}\sigma_{ij}\frac{x_i x_j}{k^2} + \text{Suku tanpa } x_i \\
    &= x_i \left( \frac{\mu_i}{k} - \frac{\gamma}{2}\frac{\sigma_i^2}{k^2} - \gamma \sum_{j \neq i}\sigma_{ij}\frac{x_j}{k^2} \right) + \text{Suku tanpa } x_i
\end{split}
\end{equation}

Definisi \textit{Exact Potential Game} mengharuskan perubahan fungsi potensial global akibat perubahan strategi satu agen sama dengan perubahan utilitas marginal agen tersebut. Kita evaluasi selisih $\Phi$ saat agen $i$ mengubah statusnya dari 1 (terpilih) menjadi 0 (tidak terpilih), dengan mengasumsikan strategi agen lain ($\vec{x}_{-i}$) tetap berbentuk
\begin{equation}
\begin{split}
    \Delta \Phi(\vec{x}) &= \Phi(1, \vec{x}_{-i}) - \Phi(0, \vec{x}_{-i}) \\
    &= 1 \cdot \left( \frac{\mu_i}{k} - \frac{\gamma}{2}\frac{\sigma_i^2}{k^2} - \gamma \sum_{j \neq i}\sigma_{ij}\frac{x_j}{k^2} \right) - 0 \\
    &= \frac{\mu_i}{k} - \frac{\gamma}{2}\frac{\sigma_i^2}{k^2} - \gamma \sum_{j \neq i}\sigma_{ij}\frac{x_j}{k^2}.
\end{split}
\end{equation}
