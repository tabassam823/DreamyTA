% Slide 21: Bab 4.3 - Pemetaan Hamiltonian Ising
\begin{frame}{Bab 4.3 - Pemetaan Hamiltonian Ising}
    \begin{itemize}
        \item \textbf{Transformasi QUBO ke Ising:} Melalui substitusi $x_i = (1-s_i)/2$.
    \end{itemize}
    \begin{equation}
        \hat{\mathcal{H}} = \sum_{i=1}^N h_i \hat{Z}_i + \sum_{i<j}^N J_{ij} \hat{Z}_i \hat{Z}_j + C
    \end{equation}
    \begin{itemize}
        \item \textbf{Parameter Fisis:}
        \begin{itemize}
            \item \textbf{Medan Lokal ($h_i$):} Menyerap bias imbal hasil individual.
            \item \textbf{Kopling ($J_{ij}$):} Menangkap korelasi risiko antar aset.
        \end{itemize}
        \item \textbf{Penalty ($A$):} Koefisien $J_{ij} \approx 0,5$ didominasi oleh pinalti anggaran guna menjaga validitas solusi.
    \end{itemize}
\end{frame}

% Slide 23: Bab 4.5 - Optimasi Berbasis Gradien (PSR)
\begin{frame}{Bab 4.5 - Optimasi Berbasis Gradien (PSR)}
    \begin{itemize}
        \item \textbf{Parameter Shift Rule (Analitik):} Memberikan gradien eksak tanpa bias melalui pergeseran fase sirkuit.
    \end{itemize}
    \begin{equation*}
        \frac{\partial E}{\partial \theta} = \frac{1}{2} \left[ E\left(\theta + \frac{\pi}{2}\right) - E\left(\theta - \frac{\pi}{2}\right) \right]
    \end{equation*}
    \begin{itemize}
        \item \textbf{Mekanisme:} Parameter $\theta$ diperbarui menggunakan arah negatif gradien untuk mencari lembah energi terendah.
        \item \textbf{Limitasi:} Membutuhkan $2p$ evaluasi sirkuit, kurang efisien untuk jumlah aset ($N$) yang besar.
    \end{itemize}
\end{frame}

% Slide 24: Bab 4.6 - Optimasi SPSA
\begin{frame}{Bab 4.6 - Optimasi SPSA}
    \begin{itemize}
        \item \textbf{Simultaneous Perturbation:} Mengestimasi gradien seluruh parameter secara simultan menggunakan vektor perturbasi stokastik.
        \item \textbf{Efisiensi Tinggi:} Hanya membutuhkan \textbf{2 evaluasi sirkuit} per iterasi ($\mathcal{O}(1)$).
        \item \textbf{Hasil Numerik (N=2):} Berhasil mengonvergensi sirkuit menuju \textit{Ground State} $|10\rangle$ dengan akurasi tinggi.
    \end{itemize}
    \begin{center}
        \textit{"SPSA memungkinkan optimasi portofolio skala besar tetap layak secara komputasi pada simulator maupun perangkat keras."}
    \end{center}
\end{frame}

% Slide 25: Bab 4.7 - Evolusi Entropi Von Neumann
\begin{frame}{Bab 4.7 - Evolusi Entropi Von Neumann}
    \begin{itemize}
        \item \textbf{Diagnostik:} Mengukur tingkat ketidakpastian sirkuit selama optimasi.
        \item \textbf{Fenomena Collapsing:}
        \begin{itemize}
            \item \textbf{Awal:} Entropi tinggi (Superposisi merata).
            \item \textbf{Akhir:} Entropi meluruh menuju nol (Wavefunction collapse ke status optimal).
        \end{itemize}
        \item \textbf{Dampak Warm-Start:} Mempercepat peluruhan entropi karena inisialisasi Nash sudah berada di wilayah probabilitas tinggi.
    \end{itemize}
\end{frame}

% Slide 26: Bab 4.8 - Analisis Kedalaman Sirkuit (Barren Plateaus)
\begin{frame}{Bab 4.8 - Analisis Kedalaman Sirkuit (Barren Plateaus)}
    \begin{itemize}
        \item \textbf{Barren Plateau:} Variansi gradien menyusut secara eksponensial terhadap jumlah qubit dan layer.
        \item \textbf{Observasi:}
        \begin{itemize}
            \item Sistem $N=2$ stabil pada kedalaman hingga $L=10$.
            \item Sistem $N=4$ menunjukkan degradasi variansi gradien yang tajam saat layer ditingkatkan.
        \end{itemize}
        \item \textbf{Solusi:} Inisialisasi EPG memandu parameter sirkuit tetap berada pada lanskap yang dapat dilatih (\textit{trainable region}).
    \end{itemize}
\end{frame}
