% Slide 4.4: Bab 2.1.4 - Teori Prospek dan Risk Aversion
\begin{frame}{Bab 2.1.4 - Teori Prospek}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Loss Aversion:} Psikologi manusia menganggap rasa sakit akibat kehilangan jauh lebih intens daripada kegembiraan dari keuntungan dengan nilai sama.
                \item Toleransi risiko dipengaruhi oleh kondisi ekonomi dan latar belakang individu.
                \begin{enumerate}
                    \item Toleransi risiko cenderung menurun seiring bertambahnya usia.
                    \item Toleransi risiko cenderung meningkat seiring meningkatnya kekayaan yang dimiliki.
                \end{enumerate}
            \end{itemize}
        \end{column}

% Slide 5.3: Bab 2.2.3 - Analisa Kuantitatif
\begin{frame}{Bab 2.2.3 - Analisa Kuantitatif}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item \textbf{Simple Return ($R$):}
                \begin{equation*}
                    R_t = \frac{P_t - P_{t-1}}{P_{t-1}}
                \end{equation*}
                \item \textbf{Logarithmic Return ($r$):}
                \begin{equation*}
                    r_t = \ln\left(\frac{P_t}{P_{t-1}}\right)
                \end{equation*}
                Dimana:
                \begin{itemize}
                    \item $P_t$: Harga pada periode waktu ke-$t$.
                    \item $P_{t-1}$: Harga pada periode waktu ke-$(t-1)$.
                    \item $\ln$: Logaritma natural.
                \end{itemize}
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{223_Price_log.png}
                \caption{Transformasi pergerakan harga saham.}

        \end{column}
    \end{columns}
\end{frame}

% Slide 5.4: Bab 2.2.4 - Model Markowitz
\begin{frame}{Bab 2.2.4 - Model Markowitz}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Mean-Variance:} merupakan model yang didasarkan oleh prinsip pareto optimum (mencari imbal hasil setinggi mungkin dengan risiko serendah mungkin).
                \begin{equation}
                    \mathcal{L}(w) = \lambda \sum_i \mu_i w_i -\sum_{i,j} \sigma_{ij} w_i w_j 
                \end{equation}
            \end{itemize}
            Dimana:
                \begin{itemize}
                    \item $\mu_i$: Imbal hasil rata-rata aset ke-$i$.
                    \item $\sigma_{ij}$: Kovarians antara aset ke-$i$ dan aset ke-$j$.
                    \item $w_i$: Proporsi bobot aset ke-$i$.
                    \item $\lambda$: Parameter bobot toleransi risiko (Semakin besar lambda, semakin besar risiko yang diambil).
                \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{224_Efficient_Frontier.png}
                \caption{Grafik Efficient Frontier (Pareto Optimum).}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}
