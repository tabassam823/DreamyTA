% Slide 4.2: Bab 2.1.2 - Inflasi dan Sejarah Uang Fiat
\begin{frame}{Bab 2.1.2 - Inflasi dan Sejarah Uang Fiat}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Nixon Shock (1971):} Presiden Richard Nixon secara sepihak mengakhiri standar emas, memulai era uang \textit{fiat} murni.
                \item \textbf{Sifat Fiat:} Nilai uang tidak lagi didukung aset fisik, melainkan kebijakan pemerintah (\textit{legal tender}).
                \item \textbf{Risiko Inflasi:} Kemampuan pencetakan uang yang tidak terbatas memicu peningkatan entropi ekonomi dan degradasi daya beli.
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{212_Inflasi.png}
                \caption{Ilustrasi siklus inflasi sistemik.}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

% Slide 4.3: Bab 2.1.3 - Investasi: Mitigasi Inflasi
\begin{frame}{Bab 2.1.3 - Investasi: Mitigasi Inflasi}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Proteksi Aset:} Investasi berfungsi sebagai instrumen untuk menekan dampak negatif inflasi pada kekayaan pribadi.
                \item \textbf{Kebutuhan Strategis:} Alokasi kapital pada aset produktif diperlukan untuk mempertahankan nilai riil aset di masa depan.
                \item \textbf{Risk-Return:} Keharusan mengambil risiko terukur untuk mendapatkan imbal hasil di atas laju inflasi.
            \end{itemize}
        \end{column}

% Slide 4.4: Bab 2.1.4 - Teori Prospek dan Risk Aversion
\begin{frame}{Bab 2.1.4 - Teori Prospek dan Risk Aversion}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Loss Aversion:} Psikologi manusia menganggap rasa sakit akibat \textbf{kehilangan} jauh lebih intens daripada kegembiraan dari keuntungan bernilai sama.
                \item \textbf{Respon Non-Linear:} Toleransi risiko agen tidak konstan, melainkan bergantung pada status kekayaan saat ini.
                \item \textbf{Evolusi Risiko:} Parameter $\gamma$ merepresentasikan kecenderungan agen menghindari volatilitas ekstrem.
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}

% Slide 5: Bab 2.2 - Strategi Analisis Investasi
\begin{frame}{Bab 2.2 - Strategi Analisis Investasi}
    \begin{itemize}
        \item \textbf{Pengantar Analisis:} Pendekatan metodologis untuk mengoptimalkan keputusan alokasi modal.
        \item \textbf{Tiga Pilar Utama:}
        \begin{enumerate}
            \item \textbf{Analisis Fundamental:} Evaluasi nilai intrinsik dan kesehatan emiten.
            \item \textbf{Analisis Teknikal:} Pemetaan psikologi pasar melalui pola visual harga.

    \end{itemize}
    \begin{center}
        \textit{"Kombinasi ketiga analisis ini membentuk landasan pengambilan keputusan yang robust."}
    \end{center}
\end{frame}

% Slide 5.3: Bab 2.2.3 - Representasi Kuantitatif Imbal Hasil
\begin{frame}{Bab 2.2.3 - Representasi Kuantitatif Imbal Hasil}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item \textbf{Simple Return ($R$):}
                \begin{equation*}
                    R_t = \frac{P_t - P_{t-1}}{P_{t-1}}
                \end{equation*}
                \item \textbf{Logarithmic Return ($r$):}
                \begin{equation*}
                    r_t = \ln\left(\frac{P_t}{P_{t-1}}\right)
                \end{equation*}
                \item \textbf{Tujuan:} Mencapai stasioneritas statistik untuk pemodelan kuantum yang stabil.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{223_Price_log.png}
                \caption{Transformasi pergerakan harga saham.}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

% Slide 5.4: Bab 2.2.4 - Optimasi Markowitz: Pareto Optimum
\begin{frame}{Bab 2.2.4 - Optimasi Markowitz: Pareto Optimum}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Mean-Variance:}
                \begin{equation*}
                    \mathcal{L}(w) = \frac{\gamma}{2} \sum_{i,j} \sigma_{ij} w_i w_j - \sum_i \mu_i w_i
                \end{equation*}
                \item \textbf{Efficient Frontier:} Himpunan portofolio \textit{Pareto-optimal} yang memberikan imbal hasil maksimal pada tingkat risiko tertentu.
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}

                \centering

                \caption{Grafik Efficient Frontier (Pareto Optimum).}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}
