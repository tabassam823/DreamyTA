% Slide 6.2: Bab 2.3.2 - Exact Potential Game (EPG)
\begin{frame}{Bab 2.3.2 - Exact Potential Game (EPG)}
    \begin{itemize}
        \item \textbf{Definisi:} Perubahan utilitas marginal pemain selaras dengan variasi fungsi potensial global $\Phi$.
        \item \textbf{Properti Utama:} Setiap peningkatan utilitas lokal berkontribusi pada penurunan energi (atau peningkatan potensial) sistem secara keseluruhan.
    \end{itemize}
    \begin{equation}
        \Phi(s_i', s_{-i}) - \Phi(s_i, s_{-i}) = u_i(s_i', s_{-i}) - u_i(s_i, s_{-i})
    \end{equation}
    \begin{itemize}
        \item \textbf{Konvergensi:} Menjamin keberadaan \textit{Pure Strategy Nash Equilibrium} (PSNE) dan stabilitas algoritma optimasi.
    \end{itemize}
\end{frame}

% Slide 6.3: Bab 2.3.3 - Markowitz sebagai EPG
\begin{frame}{Bab 2.3.3 - Markowitz sebagai EPG}
    \begin{itemize}
        \item \textbf{Pemetaan Utilitas:} Setiap aset dimodelkan sebagai pemain yang berupaya memaksimalkan utilitas (imbal hasil) dan meminimalkan risiko (interaksi kovarians).
        \item \textbf{Fungsi Potensial Portofolio:}
    \end{itemize}
    \begin{equation}
        \Phi(\omega) = \sum_i \mu_i \omega_i - \frac{\lambda}{2} \sum_{i,j} \sigma_{ij} \omega_i \omega_j
    \end{equation}
    \begin{itemize}
        \item \textbf{Sinergi:} Maksimisasi fungsi potensial $\Phi$ secara matematis ekuivalen dengan mencari konfigurasi portofolio optimal Markowitz.
    \end{itemize}
\end{frame}

% Slide 7.1: Bab 2.4.1 - Sejarah dan Evolusi Model Ising (Lenz-Ising)
\begin{frame}{Bab 2.4.1 - Sejarah dan Evolusi Model Ising (Lenz-Ising)}
    \begin{itemize}
        \item \textbf{1920-an (Konseptualisasi):} Wilhelm Lenz mengusulkan model interaksi \textit{spin} diskrit untuk menjelaskan feromagnetisme.
        \item \textbf{1925 (Solusi Ernst Ising):} Mengembangkan solusi eksak untuk sistem 1 dimensi (1D).
        \item \textbf{Kesimpulan Pesimistis:} Ising menyimpulkan tidak ada transisi fase pada 1D di atas suhu nol mutlak (fluktuasi termal mendominasi).
    \end{itemize}

    \end{center}
\end{frame}

% Slide 7.2: Bab 2.4.2 - Kebangkitan Model Ising (Onsager)
\begin{frame}{Bab 2.4.2 - Kebangkitan Model Ising (Onsager)}
    \begin{itemize}
        \item \textbf{1944 (Solusi Lars Onsager):} Merumuskan solusi eksak untuk kasus 2 dimensi (2D).
        \item \textbf{Paradigma Baru:} Membuktikan adanya transisi fase sistemik dari interaksi jarak pendek sederhana.
        \item \textbf{Ising Computing:} Transformasi model Ising menjadi bahasa standar untuk masalah optimasi kombinatorial kompleks di berbagai bidang (Biologi, Kimia, Ekonomi).
    \end{itemize}
\end{frame}

% Slide 7.3: Bab 2.4.3 - Kontribusi Mantegna dalam Ekonofisika
\begin{frame}{Bab 2.4.3 - Kontribusi Mantegna dalam Ekonofisika}
    \begin{itemize}
        \item \textbf{Rosario Mantegna (1999/2000):} Mempelopori penggunaan model Ising untuk memetakan korelasi pasar keuangan.
        \item \textbf{Penurunan Rumus:} Memetakan koefisien kopling ($J_{ij}$) dari matriks korelasi harga aset.
        \item \textbf{Kolektivitas:} Menggunakan interaksi \textit{spin-spin} untuk menjelaskan fenomena \textit{herding} dan dinamika pasar sebagai sistem partikel yang saling berinteraksi.
    \end{itemize}
    \begin{center}
        \textit{"Pasar finansial adalah sistem fisik di mana keputusan investor adalah orientasi spin yang mencari kestabilan energi."}
    \end{center}
\end{frame}

% Slide 7.4: Bab 2.4.4 - Representasi Hamiltonian Ising Modern
\begin{frame}{Bab 2.4.4 - Representasi Hamiltonian Ising Modern}
    \begin{itemize}
        \item \textbf{Representasi Aset:} Aset dianalogikan sebagai partikel \textit{spin} ($s_i \in \{-1, +1\}$).
        \item \textbf{Formulasi Hamiltonian ($\hat{H}$):}
    \end{itemize}
    \begin{equation}
        \hat{H} = - \sum_{i} h_i \hat{Z}_i - \sum_{i < j} J_{ij} \hat{Z}_i \hat{Z}_j
    \end{equation}
    \begin{itemize}
        \item \textbf{Parameter:}
        \begin{itemize}
            \item \textbf{Medan Lokal ($h_i$):} Menyerap bias imbal hasil individual.
            \item \textbf{Kopling ($J_{ij}$):} Menangkap risiko bersama (korelasi) antar aset.
        \end{itemize}
        \item \textbf{Ground State:} Solusi optimal portofolio $\equiv$ Konfigurasi energi terendah.
    \end{itemize}
\end{frame}
