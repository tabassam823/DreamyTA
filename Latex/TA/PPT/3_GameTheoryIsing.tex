% Slide 6.2: Bab 2.3.2 - Exact Potential Game (EPG)
\begin{frame}{Bab 2.3.2 - Exact Potential Game (EPG)}
    \begin{itemize}
        \item \textbf{Fungsi Potensial ($\Phi$):} Skalar yang merangkum insentif seluruh pemain.
        \item \textbf{Nash Seeker:} EPG berfungsi sebagai algoritma umum untuk mencari \textit{Nash Equilibrium} melalui maksimisasi fungsi potensial.
        \item \textbf{Karakteristik:} Perubahan utilitas lokal ($\Delta u_i$) selaras secara eksak dengan perubahan potensial global ($\Delta \Phi$).
    \end{itemize}
    \begin{equation}
        \Delta \Phi = \Phi(s_i', s_{-i}) - \Phi(s_i, s_{-i}) = u_i(s_i', s_{-i}) - u_i(s_i, s_{-i})
    \end{equation}
\end{frame}

% Slide 6.3: Bab 2.3.3 - Penurunan Markowitz ke Kerangka EPG
\begin{frame}{Bab 2.3.3 - Penurunan Markowitz ke Kerangka EPG}
    \begin{itemize}
        \item \textbf{Lagrangian Awal:}
        \begin{equation*}
            \mathcal{L}(\omega_i) = \lambda \sum_i \mu_i \omega_i - \sum_{i,j} \sigma_{ij} \omega_i \omega_j
        \end{equation*}
        \item \textbf{Transformasi Biner:} Memetakan bobot $\omega_i$ ke variabel keputusan $x_i \in \{0, 1\}$.
        \item \textbf{Fungsi Potensial EPG ($\Phi(x)$):}
        \begin{equation}
            \Phi(x) = \frac{1}{K} \sum_i \mu_i x_i - \frac{\gamma}{2K^2} \sum_{i,j} \sigma_{ij} x_i x_j
        \end{equation}
        \item \textbf{Sinergi:} Maksimisasi utilitas agen dalam game setara dengan optimasi risiko-imbal hasil portofolio.
    \end{itemize}
\end{frame}

% Slide 7.1: Bab 2.4.1 - Sejarah dan Evolusi Model Ising
\begin{frame}{Bab 2.4.1 - Sejarah dan Evolusi Model Ising}
    \begin{itemize}
        \item \textbf{Era 1920-an:} Wilhelm Lenz dan Ernst Ising merumuskan model interaksi magnetik diskrit.
        \item \textbf{Evolusi:} Dari solusi 1D yang sederhana menuju solusi 2D Onsager (1944) yang membuktikan transisi fase.
        \item \textbf{Kebangkitan Modern:} Kemajuan teknologi komputasi dan \textit{Quantum Annealing} menjadikan Ising bahasa standar optimasi kombinatorial.
    \end{itemize}

% Slide 7.2: Bab 2.4.2 - Ising dalam Ekonofisika (Mantegna)
\begin{frame}{Bab 2.4.2 - Ising dalam Ekonofisika (Mantegna)}
    \begin{itemize}
        \item \textbf{Rosario Mantegna:} Mempelopori penerapan model sistem fisis pada korelasi pasar keuangan.
        \item \textbf{Metodologi:} Menurunkan parameter interaksi sistem dari matriks korelasi aset.
        \item \textbf{Dinamika Kolektif:} Pasar dipandang sebagai kumpulan \textit{spins} yang perilakunya didorong oleh minimisasi energi sistem.
    \end{itemize}
    \begin{center}
        \textit{"Model Ising menjembatani fenomena herding investor dengan statistik fisika partikel."}
    \end{center}
\end{frame}

% Slide 7.3: Bab 2.4.3 - Formulasi Hamiltonian dari EPG
\begin{frame}{Bab 2.4.3 - Formulasi Hamiltonian dari EPG}
    \begin{itemize}
        \item \textbf{Pemetaan Energi:} Fungsi potensial EPG ($\Phi$) dipetakan menjadi operator Hamiltonian ($\hat{H} = -\Phi$).
        \item \textbf{Parameter Fisik:}
        \begin{equation}
            \hat{H} = \sum_i h_i \hat{Z}_i + \sum_{i<j} J_{ij} \hat{Z}_i \hat{Z}_j + C
        \end{equation}
        \item \textbf{Korespondensi:}
        \begin{itemize}
            \item \textbf{Medan Lokal ($h_i$):} Turunan dari ekspektasi imbal hasil (utilitas marginal).
            \item \textbf{Kopling ($J_{ij}$):} Turunan dari interaksi risiko kovariansi antar aset.
        \end{itemize}
    \end{itemize}
\end{frame}
