\begin{equation}
    x_i = \frac{s_i + 1}{2}, \quad s_i \in \{-1, 1\}
\end{equation}

\begin{equation}
    H(\vec{s}) = \sum_{i=1}^N Q_{ii} \left( \frac{s_i + 1}{2} \right) + \sum_{i \ne j} Q_{ij} \left( \frac{s_i + 1}{2} \right) \left( \frac{s_j + 1}{2} \right) + C
\end{equation}

\begin{equation}
    \left( \frac{s_i + 1}{2} \right) \left( \frac{s_j + 1}{2} \right) = \frac{1}{4} (s_i s_j + s_i + s_j + 1)
\end{equation}

\begin{equation}
    H(\vec{s}) = \sum_{i=1}^N \frac{Q_{ii}}{2} (s_i + 1) + \sum_{i \ne j} \frac{Q_{ij}}{4} (s_i s_j + s_i + s_j + 1) + C
\end{equation}

\begin{equation}
    H(\vec{s}) = \sum_{i \ne j} \frac{Q_{ij}}{4} s_i s_j + \sum_{i=1}^N \left( \frac{Q_{ii}}{2} + \sum_{j \ne i} \frac{Q_{ij}}{2} \right) s_i + \left( \sum_{i=1}^N \frac{Q_{ii}}{2} + \sum_{i \ne j} \frac{Q_{ij}}{4} + C \right)
\end{equation}

\begin{equation}
    H(\vec{s}) = \sum_{i \ne j} J_{ij} s_i s_j + \sum_{i=1}^N h_i s_i + C_{Ising}
\end{equation}

\begin{itemize}
    \item $J_{ij} = \frac{Q_{ij}}{4}$ : Kekuatan interaksi antar \textit{spin}.
    \item $h_i = \frac{Q_{ii}}{2} + \sum_{j \ne i} \frac{Q_{ij}}{2}$ : Medan magnet eksternal.
    \item $C_{Ising} = \sum_{i=1}^N \frac{Q_{ii}}{2} + \sum_{i \ne j} \frac{Q_{ij}}{4} + C$ : Pergeseran energi konstan.
\end{itemize}

\begin{equation}
    s_i \to \hat{Z}_i
\end{equation}

\begin{equation}
    \hat{Z}_i = \mathbb{I}_2 \otimes \mathbb{I}_2 \otimes \dots \otimes \sigma^z_i \otimes \dots \otimes \mathbb{I}_2
\end{equation}

\begin{equation}
    \sigma^z = \begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}, \quad \mathbb{I}_2 = \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}
\end{equation}

\begin{equation}
    \hat{H} = \sum_{i \ne j} J_{ij} \hat{Z}_i \hat{Z}_j + \sum_{i=1}^N h_i \hat{Z}_i + C_{Ising} \hat{\mathbb{I}}
\end{equation}

\begin{equation}
    E(\theta) = \langle \psi(\theta) | \hat{H} | \psi(\theta) \rangle
\end{equation}

\begin{equation}
    E(\theta) = \sum_{i \ne j} J_{ij} \langle \psi(\theta) | \hat{Z}_i \hat{Z}_j | \psi(\theta) \rangle + \sum_{i=1}^N h_i \langle \psi(\theta) | \hat{Z}_i | \psi(\theta) \rangle + C_{Ising}
\end{equation}
