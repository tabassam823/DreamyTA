\begin{equation}
    L(\vec{\omega}) = \vec{\omega}^T \Sigma \vec{\omega} - \lambda \mu^T \vec{\omega}
\end{equation}
\begin{itemize}
    \item $\vec{\omega}$: Vektor bobot aset.
    \item $\Sigma$: Matriks kovariansi dengan elemen $\sigma_{ij}$.
    \item $\mu$: Matriks \textit{expected return} dengan elemen $\mu_i$.
    \item $\lambda$: Parameter \textit{risk tolerance}.
\end{itemize}

\begin{equation}
    \lambda = \frac{2}{\gamma} \implies \frac{1}{\lambda} = \frac{\gamma}{2}
\end{equation}
\begin{itemize}
    \item $\gamma$: Parameter \textit{risk aversion}.
\end{itemize}

\begin{equation}
    \vec{\omega} = \frac{\vec{x}}{k}, \quad k \in \mathbb{R}
\end{equation}
\begin{itemize}
    \item $\vec{x}$: Vektor strategi dalam \textit{game}.
    \item $k$: Parameter skalar (mengingat $\sum_i \omega_i = \sum_i \frac{x_i}{k}$).
\end{itemize}

\begin{align}
    L(\vec{x}) &= \left( \frac{\vec{x}}{k} \right)^T \Sigma \left( \frac{\vec{x}}{k} \right) - \lambda \mu^T \left( \frac{\vec{x}}{k} \right) \\
    L(\vec{x}) &= \frac{1}{k^2} \vec{x}^T \Sigma \vec{x} - \frac{\lambda}{k} \mu^T \vec{x}
\end{align}

\begin{align}
    \vec{x}^T \Sigma \vec{x} &= \sum_{i=1}^N \sum_{j=1}^N \sigma_{ij} x_i x_j \\
    \mu^T \vec{x} &= \sum_{i=1}^N \mu_i x_i
\end{align}

\begin{align}
    U(\vec{x}) &= L(\vec{x}) \cdot \left( -\frac{1}{\lambda} \right) \\
    U(\vec{x}) &= \left( \frac{1}{k^2} \sum_{i=1}^N \sum_{j=1}^N \sigma_{ij} x_i x_j - \frac{\lambda}{k} \sum_{i=1}^N \mu_i x_i \right) \cdot \left( -\frac{\gamma}{2} \right) \\
    U(\vec{x}) &= \frac{1}{k} \sum_{i=1}^N \mu_i x_i - \frac{\gamma}{2k^2} \sum_{i=1}^N \sum_{j=1}^N \sigma_{ij} x_i x_j
\end{align}
\begin{itemize}
    \item $U(\vec{x})$: Fungsi potensial \textit{Exact Potential Game}.
\end{itemize}

\begin{align}
    u_i(x_i, \vec{x}_{-i}) &= U(x_i, \vec{x}_{-i}) - U(0, \vec{x}_{-i}) \\
    u_i(x_i, \vec{x}_{-i}) &= x_i \left( \frac{\mu_i}{k} - \frac{\gamma \sigma_{ii} x_i}{2k^2} - \frac{\gamma}{k^2} \sum_{j \ne i} \sigma_{ij} x_j \right)
\end{align}

Skenario A: $x_i = 0 \to x_i' = 1$
\begin{equation}
    \Delta u_i(0 \to 1) = \frac{\mu_i}{k} - \frac{\gamma \sigma_{ii}}{2k^2} - \frac{\gamma}{k^2} \sum_{j \ne i} \sigma_{ij} x_j
\end{equation}

Skenario B: $x_i = 1 \to x_i' = 0$
\begin{equation}
    \Delta u_i(1 \to 0) = -\frac{\mu_i}{k} + \frac{\gamma \sigma_{ii}}{2k^2} + \frac{\gamma}{k^2} \sum_{j \ne i} \sigma_{ij} x_j
\end{equation}
\begin{itemize}
    \item $\Delta u_i$: Perubahan utilitas marginal pemain $i$.
\end{itemize}

\text{Kasus Khusus ($k=1$):}
\begin{equation}
    \Delta u_i |_{k=1} = \mu_i - \frac{\gamma}{2} \sigma_{ii} - \gamma \sum_{j \ne i} \sigma_{ij} x_j
\end{equation}
