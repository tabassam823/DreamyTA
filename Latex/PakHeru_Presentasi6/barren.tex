\begin{equation}
    \mathbb{E}_{\theta} \left[ \frac{\partial E(\theta)}{\partial \theta_k} \right] = 0
\end{equation}
$\partial_k E(\theta)$ : Gradien fungsi energi terhadap parameter $\theta_k$.

\begin{equation}
    \text{Var}_{\theta} \left[ \partial_k E(\theta) \right] = \mathbb{E}_{\theta} \left[ \left( \partial_k E(\theta) \right)^2 \right] - \left( \mathbb{E}_{\theta} \left[ \partial_k E(\theta) \right] \right)^2
\end{equation}
$\text{Var}_{\theta}$ : Varians gradien terhadap distribusi parameter $\theta$.

\begin{equation}
    \text{Var}_{\theta} \left[ \partial_k E(\theta) \right] = \mathbb{E}_{\theta} \left[ \left( \partial_k E(\theta) \right)^2 \right]
\end{equation}

Dekomposisi ansatz $\hat{U}(\theta)$ menjadi matriks uniter pra parameter $\hat{U}_A$ dan pasca parameter $\hat{U}_B$
\begin{equation}
    E(\theta) = \text{Tr}\left[ \hat{H} \hat{U}_B e^{-i \theta_k \hat{V}_k} \hat{U}_A |0\rangle \langle 0| \hat{U}_A^\dagger e^{i \theta_k \hat{V}_k} \hat{U}_B^\dagger \right]
\end{equation}

\begin{equation}
    \left(\partial_k E(\theta)\right)^2 = \left( \frac{-i}{2} \text{Tr} \left[\hat{H} \hat{U}_B \left[\hat{V}_k \hat{U}_A |0\rangle \langle 0| \hat{U}_A^\dagger\right] \hat{V}_k\right] \right)^2
\end{equation}

Integrasi ruang hilbert dengan ukuran Haal orde-2 pada matriks uniter acak:
\begin{equation}
    \int_{\mathcal{U}} f(U) d\mu(U)
\end{equation}
$d\mu(U)$ : Ukuran Haar (\textit{Haar measure}) pada grup uniter $\mathcal{U}$.

\begin{equation}
    \int_{\mathcal{U}(N)} U A U^\dagger B U C U^\dagger dU = \frac{\text{Tr}[A]\text{Tr}[C]}{N^2-1} B + \frac{\text{Tr}[AC]}{N^2-1} \text{Tr}[B] I - \frac{\text{Tr}[AC]\text{Tr}[B]}{N(N^2-1)} I - \frac{\text{Tr}[A]\text{Tr}[C]}{N(N^2-1)} B
\end{equation}

\begin{equation}
    \int_{\mathcal{U}(N)} |\text{Tr}[A U B U^\dagger]|^2 dU = \frac{\text{Tr}[A^2]\text{Tr}[B^2] + |\text{Tr}[A]|^2|\text{Tr}[B]|^2 - \frac{1}{N}(\text{Tr}[A^2]|\text{Tr}[B]|^2 + \text{Tr}[B^2]|\text{Tr}[A]|^2)}{N^2-1}
\end{equation}

\begin{equation}
    \int_{\mathcal{U}(N)} |\text{Tr}[A U B U^\dagger]|^2 dU \approx \frac{\text{Tr}[A^2] \text{Tr}[B^2]}{N^2 - 1}
\end{equation}

\begin{equation}
    \int_{U(N)} U_{i,j} U_{k,l} U^*_{m,n} U^*_{p,q} d\mu(U) = \frac{\delta_{im}\delta_{jn}\delta_{kp}\delta_{lq} + \delta_{ip}\delta_{jq}\delta_{km}\delta_{ln}}{N^2 - 1} - \frac{\delta_{im}\delta_{jq}\delta_{kp}\delta_{ln} + \delta_{ip}\delta_{jn}\delta_{km}\delta_{lq}}{N(N^2 - 1)}
\end{equation}
$N = 2^n$ : Dimensi ruang Hilbert untuk $n$ qubit.

\begin{equation}
    \text{Var}_{\theta} \left[ \partial_k E(\theta) \right] \sim \frac{1}{2^{2n}}
\end{equation}

\begin{equation}
    \text{Var}_{\theta} \left[ \partial_k E(\theta) \right] = \int_{U_A, U_B} \left( \partial_k \langle 0 | U_B^\dagger U_A^\dagger \hat{H} U_A U_B | 0 \rangle \right)^2 d\mu(U_A) d\mu(U_B)
\end{equation}
$U_A, U_B$ : Matriks uniter acak (\textit{random unitaries}) yang membentuk \textit{2-design}.
