\section{APPENDIX B}

\subsection{Paragraf 1: Estimasi Galat Total}
\begin{frame}{Parameter $\delta$}
    \begin{itemize}
        \item \textbf{Batas Atas:} $\delta$ didefinisikan sebagai \textit{upper bound} akumulasi seluruh kesalahan eksperimental.
        \item \textbf{Percampuran Galat:} Kesulitan teknis memisahkan derau gerbang, galat pengukuran, dan limitasi sistem.
        \item \textbf{Skenario Terburuk:} Rumusan estimasi galat total dari observasi parsial skenario terburuk (penjumlahan linear).
    \end{itemize}
\end{frame}

\begin{frame}{Pertanyaan yang dijawab pada Paragraf 1}
    \centering
    \small
    \begin{tabular}{lp{4.5cm}p{6cm}}
        \toprule
        Kalimat ke & Pertanyaan & Jawaban \\
        \midrule
        1 & Apa itu parameter $\delta$? & Batas atas (\textit{upper bound}) akumulasi galat. \\
        2 & Mengapa sulit menentukan sumber? & Adanya percampuran derau gerbang dan pengukuran. \\
        3 & Strategi estimasi? & Penjumlahan linear galat (\textit{worst-case scenario}). \\
        \bottomrule
    \end{tabular}
\end{frame}

\subsection{Paragraf 2: Kalibrasi dan Ekstrapolasi}
\begin{frame}{Metodologi Galat}
    \begin{itemize}
        \item \textbf{Langkah Awal:} Pengukuran fidelitas pada kasus $2 \times 2$ sebagai basis kalibrasi.
        \item \textbf{Galat per Gerbang:} Estimasi deviasi hasil eksperimen terhadap teoretis ideal per-interaksi 2-qubit.
        \item \textbf{Ekstrapolasi 4x4:} Prediksi perilaku kesalahan sistem $4 \times 4$ berdasarkan jumlah gerbang sirkuit.
    \end{itemize}
\end{frame}

\begin{frame}{Pertanyaan yang dijawab pada Paragraf 2}
    \centering
    \small
    \begin{tabular}{lp{4.5cm}p{6cm}}
        \toprule
        Kalimat ke & Pertanyaan & Jawaban \\
        \midrule
        1 & Kalibrasi galat mulai? & Hitung fidelitas sirkuit pada matriks $2 \times 2$. \\
        3 & Hubungan galat per gate? & $\delta$ = (galat gate) $\times$ (jumlah gate dalam sirkuit). \\
        4 & Estimasi kasus besar? & Proyeksi akumulasi galat via jumlah gate sirkuit $4 \times 4$. \\
        \bottomrule
    \end{tabular}
\end{frame}
